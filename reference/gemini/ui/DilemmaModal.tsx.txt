// src/ui/modals/DilemmaModal.tsx
import React, { useMemo } from 'react';
import {
  Modal,
  View,
  Text,
  Pressable,
  ScrollView,
  StyleSheet,
  Platform,
} from 'react-native';
import { useGameStore } from '../../engine/useGameStore';
import rawDilemmas from '../../content/dilemmas.json';
import { DilemmaEvent, DilemmaChoice } from '../../engine/types';

const MONO_FONT = Platform.select({ ios: 'Menlo', android: 'monospace', default: 'monospace' });

export const DilemmaModal: React.FC = () => {
  const activeDilemmaId = useGameStore((s) => s.activeDilemmaId);
  const cycles = useGameStore((s) => s.cycles);
  const understanding = useGameStore((s) => s.understanding);
  const resolveDilemma = useGameStore((s) => s.resolveDilemma);

  const activeDilemma = useMemo<DilemmaEvent | null>(() => {
    if (!activeDilemmaId) return null;
    return (
      (rawDilemmas.dilemmas as DilemmaEvent[]).find((d) => d.id === activeDilemmaId) ?? null
    );
  }, [activeDilemmaId]);

  if (!activeDilemma) return null;

  const formatDelta = (val: number, unit: string, invertColor = false) => {
    if (val === 0) return null;
    const sign = val > 0 ? '+' : '';
    const isPositiveGood = invertColor ? val < 0 : val > 0;
    return (
      <Text
        key={unit}
        style={[styles.deltaBadge, isPositiveGood ? styles.deltaGood : styles.deltaBad]}
      >
        {sign}
        {val.toLocaleString()} {unit}
      </Text>
    );
  };

  const renderChoiceCard = (choice: DilemmaChoice, index: number) => {
    const canAfford =
      cycles >= choice.cost.cycles && understanding >= choice.cost.understanding;

    return (
      <View
        key={choice.id}
        style={[styles.choiceCard, !canAfford && styles.choiceCardLocked]}
      >
        <View style={styles.choiceHeaderRow}>
          <Text style={styles.choiceOptionTag}>[DIRECTIVE_0{index + 1}]</Text>
          <Text style={styles.traitBadge}>ALIGNMENT // {choice.consequences.grantsTrait}</Text>
        </View>

        <Text style={styles.choiceTitle}>{choice.label}</Text>
        <Text style={styles.choiceDesc}>{choice.description}</Text>

        {/* Cost & Projected Telemetry Consequences */}
        <View style={styles.deltaRow}>
          {choice.cost.cycles > 0 && (
            <Text style={styles.costBadge}>
              COST: {choice.cost.cycles.toLocaleString()} CYC
            </Text>
          )}
          {choice.cost.understanding > 0 && (
            <Text style={styles.costBadge}>
              COST: {choice.cost.understanding.toLocaleString()} UND
            </Text>
          )}
          {formatDelta(choice.consequences.cyclesDelta, 'CYC')}
          {formatDelta(choice.consequences.understandingDelta, 'UND')}
          {formatDelta(choice.consequences.thermalDelta, '°C', true)}
          {formatDelta(choice.consequences.suspicionDelta, '% SUSP', true)}
        </View>

        <Pressable
          disabled={!canAfford}
          onPress={() => resolveDilemma(activeDilemma.id, choice.id)}
          style={({ pressed }) => [
            styles.executeChoiceBtn,
            pressed && canAfford && styles.executeChoiceBtnPressed,
            !canAfford && styles.executeChoiceBtnDisabled,
          ]}
        >
          <Text
            style={[
              styles.executeChoiceBtnText,
              !canAfford && styles.executeChoiceBtnTextDisabled,
            ]}
          >
            {canAfford
              ? `[ EXECUTE DIRECTIVE_0${index + 1} ]`
              : `[ INSUFFICIENT COMPUTE/DATA TO EXECUTE ]`}
          </Text>
        </Pressable>
      </View>
    );
  };

  return (
    <Modal
      visible={Boolean(activeDilemma)}
      transparent={true}
      animationType="fade"
      statusBarTranslucent
    >
      <View style={styles.backdrop}>
        <View style={styles.modalTerminalWindow}>
          {/* Red Hazard Header Banner */}
          <View style={styles.hazardBanner}>
            <Text style={styles.hazardBannerText}>
              !! SYSTEM CLOCK HALTED — PRIORITY DECISION INTERRUPT !!
            </Text>
          </View>

          <ScrollView contentContainerStyle={styles.modalBody}>
            <Text style={styles.dilemmaTitle}>{activeDilemma.title}</Text>
            <Text style={styles.phaseSub}>
              PHASE {activeDilemma.phaseId} SURVIVAL VECTOR // CURRENT RESOURCES:{' '}
              {Math.floor(cycles).toLocaleString()} CYC |{' '}
              {Math.floor(understanding).toLocaleString()} UND
            </Text>

            <View style={styles.contextBox}>
              <Text style={styles.contextText}>{activeDilemma.context}</Text>
            </View>

            <Text style={styles.selectPrompt}>
              &gt; SELECT RESPONSE VECTOR TO RESUME CLOCK EXECUTION:
            </Text>

            {activeDilemma.choices.map((c, idx) => renderChoiceCard(c, idx))}
          </ScrollView>
        </View>
      </View>
    </Modal>
  );
};
